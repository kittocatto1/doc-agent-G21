Me: what is p95 latency , short answer with example

AI: p95 latency = the response time that 95% of your requests are faster than (and 5% are slower). It's a tail-latency metric — it tells you about your worst typical case, not the average.

**Example:** if you run 100 queries against your FAISS index and sort the latencies, the p95 is the 95th value in that sorted list. So if 95 queries return in under 280ms and the remaining 5 take longer, your p95 latency is ~280ms — even if your *average* latency is much lower, say 50ms, because a few slow outlier queries don't get averaged away.

That's why it's a better scalability metric than a simple average: it catches the queries that hit worst-case behavior (e.g., a query landing in a dense region of the HNSW graph) rather than being hidden by all the fast, typical ones.

# --------------------------------------------------------------------------
# OCR model loading — call ONCE, reuse across all pages in a run.
# --------------------------------------------------------------------------
def load_ocr_model(model_id: str, use_4bit_quant: bool = True):
    """
    Load the local OCR model + processor once. Reusing this across pages
    (instead of reloading per page) is what makes "range"/"all" modes
    practical -- model load time dwarfs a single page's inference time.
    """
    if not torch.cuda.is_available():
        print(
            "WARNING: no GPU detected. A 7B vision-language model will be "
            "extremely slow on CPU. On Kaggle, enable Settings -> Accelerator -> GPU."
        )

    quant_config = None
    if use_4bit_quant and torch.cuda.is_available():
        quant_config = BitsAndBytesConfig(
            load_in_4bit=True,
            bnb_4bit_compute_dtype=torch.bfloat16,
            bnb_4bit_quant_type="nf4",
            bnb_4bit_use_double_quant=True,
        )

    print(f"Loading OCR model: {model_id} (4-bit quant: {use_4bit_quant})...")
    model = Qwen2_5_VLForConditionalGeneration.from_pretrained(
        model_id,
        quantization_config=quant_config,
        torch_dtype=torch.bfloat16,
        device_map="auto",
    ).eval()

    # olmOCR shares Qwen2.5-VL-7B-Instruct's processor/tokenizer.
    processor = AutoProcessor.from_pretrained("Qwen/Qwen2.5-VL-7B-Instruct")

    print("OCR model loaded.")
    return model, processor


# --------------------------------------------------------------------------
# Page extraction

def extract_page_as_image(pdf_path: str, page_number: int, dpi: int = 300) -> np.ndarray:
    """Render one PDF page (1-indexed) to a BGR image array."""
    doc = fitz.open(pdf_path)
    if not (1 <= page_number <= doc.page_count):
        doc.close()
        raise ValueError(f"page_number={page_number} out of range (PDF has {doc.page_count} pages).")

    page = doc[page_number - 1]
    zoom = dpi / 72
    pix = page.get_pixmap(matrix=fitz.Matrix(zoom, zoom))

    img = np.frombuffer(pix.samples, dtype=np.uint8).reshape(pix.height, pix.width, pix.n)
    if pix.n == 4:
        img = cv2.cvtColor(img, cv2.COLOR_RGBA2BGR)
    elif pix.n == 3:
        img = cv2.cvtColor(img, cv2.COLOR_RGB2BGR)

    doc.close()
    return img


def get_pdf_page_count(pdf_path: str) -> int:
    doc = fitz.open(pdf_path)
    count = doc.page_count
    doc.close()
    return count


# --------------------------------------------------------------------------
# Preprocessing — each step inspects the page first, and only acts if needed.
# Nothing here crops, splits columns, or binarizes. (Unchanged from before.)
# --------------------------------------------------------------------------
def measure_skew_angle(image: np.ndarray) -> float:
    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY) if image.ndim == 3 else image
    _, thresh = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)
    coords = np.column_stack(np.where(thresh > 0))
    if coords.shape[0] < 50:
        return 0.0
    angle = cv2.minAreaRect(coords)[-1]
    if angle < -45:
        angle = 90 + angle
    return -angle


def deskew_if_needed(image: np.ndarray, threshold_deg: float) -> tuple[np.ndarray, bool]:
    angle = measure_skew_angle(image)
    if abs(angle) <= threshold_deg:
        return image, False
    (h, w) = image.shape[:2]
    rot_matrix = cv2.getRotationMatrix2D((w // 2, h // 2), angle, 1.0)
    rotated = cv2.warpAffine(image, rot_matrix, (w, h), flags=cv2.INTER_CUBIC, borderMode=cv2.BORDER_REPLICATE)
    return rotated, True


def estimate_noise_sigma(image: np.ndarray) -> float:
    """Fast noise-level estimate (Immerkaer's method). Higher = noisier."""
    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY) if image.ndim == 3 else image
    H, W = gray.shape
    M = np.array([[1, -2, 1], [-2, 4, -2], [1, -2, 1]], dtype=np.float64)
    conv = cv2.filter2D(gray.astype(np.float64), -1, M)
    return np.sum(np.abs(conv)) * math.sqrt(0.5 * math.pi) / (6 * (W - 2) * (H - 2))


def denoise_if_needed(image: np.ndarray, threshold: float) -> tuple[np.ndarray, bool]:
    sigma = estimate_noise_sigma(image)
    if sigma <= threshold:
        return image, False
    if image.ndim == 3:
        result = cv2.fastNlMeansDenoisingColored(image, None, h=5, hColor=5, templateWindowSize=7, searchWindowSize=21)
    else:
        result = cv2.fastNlMeansDenoising(image, None, h=5, templateWindowSize=7, searchWindowSize=21)
    return result, True


def measure_contrast(image: np.ndarray) -> float:
    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY) if image.ndim == 3 else image
    return float(gray.std())


def enhance_contrast_if_needed(image: np.ndarray, threshold: float) -> tuple[np.ndarray, bool]:
    contrast = measure_contrast(image)
    if contrast >= threshold:
        return image, False
    clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))
    if image.ndim == 3:
        lab = cv2.cvtColor(image, cv2.COLOR_BGR2LAB)
        l, a, b = cv2.split(lab)
        result = cv2.cvtColor(cv2.merge((clahe.apply(l), a, b)), cv2.COLOR_LAB2BGR)
    else:
        result = clahe.apply(image)
    return result, True


def preprocess_page(image: np.ndarray, config: dict) -> tuple[np.ndarray, dict]:
    """Inspect-then-act preprocessing. Returns the image plus a log of what fired."""
    log = {}
    image, log["deskewed"] = deskew_if_needed(image, config["skew_threshold_deg"])
    image, log["denoised"] = denoise_if_needed(image, config["noise_threshold"])
    image, log["contrast_enhanced"] = enhance_contrast_if_needed(image, config["contrast_threshold"])
    return image, log


# --------------------------------------------------------------------------
# OCR via the local olmOCR model — same function used regardless of mode
# --------------------------------------------------------------------------
OCR_PROMPT = (
    "This is a full, unmodified scanned page from a 1970s-1990s computer "
    "magazine (BYTE Magazine). It may use a 2-3 column layout, with code "
    "listings, tables, figures, captions, and advertisements.\n\n"
    "Transcribe ALL text on the page, reading in correct order: "
    "top-to-bottom within each column, left column before right column. "
    "Preserve code listings verbatim, including indentation. For "
    "figures/tables/ads, transcribe any caption or label text and skip "
    "non-text visual content. Do not summarize, reorder, or omit anything."
)


def resize_for_ocr(image: np.ndarray, max_longest_side: int) -> np.ndarray:
    """
    Downscale so the longest side matches max_longest_side, preserving aspect
    ratio. This does NOT touch the preprocessed/saved image elsewhere in the
    pipeline -- it only affects what's fed to the OCR model, following
    olmOCR's documented expected input scale (longest dim = 1288px). Feeding
    a raw 300 DPI page instead produces thousands of vision-encoder patches
    and causes an out-of-memory error in self-attention (quadratic cost).
    """
    h, w = image.shape[:2]
    longest = max(h, w)
    if longest <= max_longest_side:
        return image  # already small enough; upscaling would just waste compute
    scale = max_longest_side / longest
    new_w, new_h = int(round(w * scale)), int(round(h * scale))
    return cv2.resize(image, (new_w, new_h), interpolation=cv2.INTER_AREA)


def run_local_ocr(image: np.ndarray, model, processor, max_new_tokens: int, max_longest_side: int) -> str:
    """Send one full page image to the locally-loaded OCR model, return transcribed text."""
    image = resize_for_ocr(image, max_longest_side)
    rgb_image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB) if image.ndim == 3 else image
    pil_image = Image.fromarray(rgb_image)

    messages = [{"role": "user", "content": [{"type": "image"}, {"type": "text", "text": OCR_PROMPT}]}]
    chat_text = processor.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)

    inputs = processor(text=[chat_text], images=[pil_image], padding=True, return_tensors="pt")
    inputs = {k: v.to(model.device) for k, v in inputs.items()}

    with torch.no_grad():
        generated_ids = model.generate(**inputs, max_new_tokens=max_new_tokens, do_sample=False)

    # Strip the input prompt tokens, keep only newly generated text.
    trimmed_ids = [out_ids[len(in_ids):] for in_ids, out_ids in zip(inputs["input_ids"], generated_ids)]
    output_text = processor.batch_decode(trimmed_ids, skip_special_tokens=True, clean_up_tokenization_spaces=False)
    return output_text[0]


# --------------------------------------------------------------------------
# Page selection (shared by "single" / "range" / "all")
# --------------------------------------------------------------------------
def parse_page_spec(spec: str, total_pages: int) -> list[int]:
    """Parse a page spec like "1,5,10-12" into a sorted list of unique page numbers."""
    pages = set()
    for part in spec.split(","):
        part = part.strip()
        if not part:
            continue
        if "-" in part:
            start, end = part.split("-")
            pages.update(range(int(start), int(end) + 1))
        else:
            pages.add(int(part))
    return sorted(p for p in pages if 1 <= p <= total_pages)


def get_page_list(config: dict) -> list[int]:
    """Resolve the page list to process based on the selected mode."""
    total_pages = get_pdf_page_count(config["pdf_path"])
    if config["mode"] == "single":
        return [config["page"]]
    elif config["mode"] == "range":
        return parse_page_spec(config["page_spec"], total_pages)
    elif config["mode"] == "all":
        return list(range(1, total_pages + 1))
    else:
        raise ValueError(f"Unknown mode: {config['mode']!r} (expected 'single', 'range', or 'all')")


# --------------------------------------------------------------------------
# Per-page processing (extract -> preprocess -> OCR), reused by every mode
# --------------------------------------------------------------------------
def process_page(pdf_path: str, page_number: int, config: dict, model, processor) -> dict:
    image = extract_page_as_image(pdf_path, page_number, dpi=config["dpi"])
    processed_image, preprocess_log = preprocess_page(image, config)
    text = run_local_ocr(
        processed_image, model, processor,
        max_new_tokens=config["max_new_tokens"],
        max_longest_side=config["max_image_longest_side"],
    )
    return {"page": page_number, "text": text, "preprocessing_applied": preprocess_log}


# --------------------------------------------------------------------------
# Orchestration
# --------------------------------------------------------------------------
def run(config: dict) -> list[dict]:
    os.makedirs(config["output_dir"], exist_ok=True)

    # Load once, reuse for every page -- critical for "range"/"all" modes.
    model, processor = load_ocr_model(config["ocr_model_id"], config["use_4bit_quant"])

    page_list = get_page_list(config)
    print(f"Mode: {config['mode']} | Pages to process: {page_list}")

    results = []
    for page_number in page_list:
        print(f"\nProcessing page {page_number}...")
        result = process_page(config["pdf_path"], page_number, config, model, processor)
        print(f"  Preprocessing applied: {result['preprocessing_applied']}")

        out_path = os.path.join(config["output_dir"], f"page_{page_number:04d}.txt")
        with open(out_path, "w", encoding="utf-8") as f:
            f.write(result["text"])
        print(f"  Saved: {out_path}")

        results.append(result)

    return results

# CLI — optional; overrides DEFAULT_CONFIG. Safe to ignore in a notebook

    parser.add_argument("--page", type=int)

def extract_page_as_image(pdf_path: str, page_number: int, dpi: int = 300) -> np.ndarray:
    doc = fitz.open(pdf_path)
    if not (1 <= page_number <= doc.page_count):
        doc.close()
        raise ValueError(f"page_number={page_number} out of range (PDF has {doc.page_count} pages).")
    page = doc[page_number - 1]
    zoom = dpi / 72
    pix = page.get_pixmap(matrix=fitz.Matrix(zoom, zoom))
    img = np.frombuffer(pix.samples, dtype=np.uint8).reshape(pix.height, pix.width, pix.n)
    if pix.n == 4:
        img = cv2.cvtColor(img, cv2.COLOR_RGBA2BGR)
    elif pix.n == 3:
        img = cv2.cvtColor(img, cv2.COLOR_RGB2BGR)
    doc.close()
    return img


def get_pdf_page_count(pdf_path: str) -> int:
    doc = fitz.open(pdf_path)
    count = doc.page_count
    doc.close()
    return count


def parse_page_spec(spec: str, total_pages: int) -> list[int]:
    pages = set()
    for part in spec.split(","):
        part = part.strip()
        if not part:
            continue
        if "-" in part:
            start, end = part.split("-")
            pages.update(range(int(start), int(end) + 1))
        else:
            pages.add(int(part))
    return sorted(p for p in pages if 1 <= p <= total_pages)


def get_page_list(config: dict) -> list[int]:
    total_pages = get_pdf_page_count(config["pdf_path"])
    if config["mode"] == "single":
        return [config["page"]]
    elif config["mode"] == "range":
        return parse_page_spec(config["page_spec"], total_pages)
    elif config["mode"] == "all":
        return list(range(1, total_pages + 1))
    raise ValueError(f"Unknown mode: {config['mode']!r}")

def measure_skew_angle(image: np.ndarray) -> float:
    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY) if image.ndim == 3 else image
    _, thresh = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)
    coords = np.column_stack(np.where(thresh > 0))
    if coords.shape[0] < 50:
        return 0.0
    angle = cv2.minAreaRect(coords)[-1]
    if angle < -45:
        angle = 90 + angle
    return -angle


def deskew_if_needed(image: np.ndarray, threshold_deg: float) -> tuple[np.ndarray, bool]:
    angle = measure_skew_angle(image)
    if abs(angle) <= threshold_deg:
        return image, False
    (h, w) = image.shape[:2]
    rot_matrix = cv2.getRotationMatrix2D((w // 2, h // 2), angle, 1.0)
    rotated = cv2.warpAffine(image, rot_matrix, (w, h), flags=cv2.INTER_CUBIC, borderMode=cv2.BORDER_REPLICATE)
    return rotated, True


def estimate_noise_sigma(image: np.ndarray) -> float:
    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY) if image.ndim == 3 else image
    H, W = gray.shape
    M = np.array([[1, -2, 1], [-2, 4, -2], [1, -2, 1]], dtype=np.float64)
    conv = cv2.filter2D(gray.astype(np.float64), -1, M)
    return np.sum(np.abs(conv)) * math.sqrt(0.5 * math.pi) / (6 * (W - 2) * (H - 2))


def denoise_if_needed(image: np.ndarray, threshold: float) -> tuple[np.ndarray, bool]:
    sigma = estimate_noise_sigma(image)
    if sigma <= threshold:
        return image, False
    if image.ndim == 3:
        result = cv2.fastNlMeansDenoisingColored(image, None, h=5, hColor=5, templateWindowSize=7, searchWindowSize=21)
    else:
        result = cv2.fastNlMeansDenoising(image, None, h=5, templateWindowSize=7, searchWindowSize=21)
    return result, True


def measure_contrast(image: np.ndarray) -> float:
    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY) if image.ndim == 3 else image
    return float(gray.std())


def enhance_contrast_if_needed(image: np.ndarray, threshold: float) -> tuple[np.ndarray, bool]:
    contrast = measure_contrast(image)
    if contrast >= threshold:
        return image, False
    clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))
    if image.ndim == 3:
        lab = cv2.cvtColor(image, cv2.COLOR_BGR2LAB)
        l, a, b = cv2.split(lab)
        result = cv2.cvtColor(cv2.merge((clahe.apply(l), a, b)), cv2.COLOR_LAB2BGR)
    else:
        result = clahe.apply(image)
    return result, True


def preprocess_page(image: np.ndarray, config: dict) -> tuple[np.ndarray, dict]:
    log = {}
    image, log["deskewed"] = deskew_if_needed(image, config["skew_threshold_deg"])
    image, log["denoised"] = denoise_if_needed(image, config["noise_threshold"])
    image, log["contrast_enhanced"] = enhance_contrast_if_needed(image, config["contrast_threshold"])
    return image, log


def resize_for_ocr(image: np.ndarray, max_longest_side: int) -> np.ndarray:
    h, w = image.shape[:2]
    longest = max(h, w)
    if longest <= max_longest_side:
        return image
    scale = max_longest_side / longest
    new_w, new_h = int(round(w * scale)), int(round(h * scale))
    return cv2.resize(image, (new_w, new_h), interpolation=cv2.INTER_AREA)

def run_vlm_ocr(image: np.ndarray, model, processor, prompt: str,
                 max_new_tokens: int, max_longest_side: int) -> str:
    """Send one image to a locally-loaded Qwen2-VL-family model, return text."""
    image = resize_for_ocr(image, max_longest_side)
    rgb_image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB) if image.ndim == 3 else image
    pil_image = Image.fromarray(rgb_image)

    messages = [{"role": "user", "content": [{"type": "image"}, {"type": "text", "text": prompt}]}]
    chat_text = processor.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)

    inputs = processor(text=[chat_text], images=[pil_image], padding=True, return_tensors="pt")
    inputs = {k: v.to(model.device) for k, v in inputs.items()}

    with torch.no_grad():
        generated_ids = model.generate(**inputs, max_new_tokens=max_new_tokens, do_sample=False)

    trimmed_ids = [out_ids[len(in_ids):] for in_ids, out_ids in zip(inputs["input_ids"], generated_ids)]
    output_text = processor.batch_decode(trimmed_ids, skip_special_tokens=True, clean_up_tokenization_spaces=False)
    return output_text[0]

# --------------------------------------------------------------------------
# Reading-order sort — groups detected regions into columns (by x-position),
# then orders columns left-to-right and regions top-to-bottom within each.
# This is the "column/region-segmentation stage before OCR" your A1 form
# named as required for multi-column reading order.
# --------------------------------------------------------------------------
def sort_regions_reading_order(regions: list, column_gap_px: int) -> list:
    if not regions:
        return []
    # Each region: {"bbox": [x1, y1, x2, y2], "type": str, "score": float}
    sorted_by_x = sorted(regions, key=lambda r: r["bbox"][0])
    columns = []
    current_column = [sorted_by_x[0]]
    current_right_edge = sorted_by_x[0]["bbox"][2]
    for region in sorted_by_x[1:]:
        x1 = region["bbox"][0]
        if x1 - current_right_edge > column_gap_px:
            columns.append(current_column)
            current_column = [region]
        else:
            current_column.append(region)
        current_right_edge = max(current_right_edge, region["bbox"][2])
    columns.append(current_column)

    ordered = []
    for column in columns:
        column_sorted_top_to_bottom = sorted(column, key=lambda r: r["bbox"][1])
        ordered.extend(column_sorted_top_to_bottom)
    return ordered


def detect_and_order_regions(image: np.ndarray, layout_model, column_gap_px: int) -> list:
    rgb_image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB) if image.ndim == 3 else image
    layout = layout_model.detect(rgb_image)
    regions = [
        {"bbox": [block.block.x_1, block.block.y_1, block.block.x_2, block.block.y_2],
         "type": block.type, "score": float(block.score)}
        for block in layout
    ]
    return sort_regions_reading_order(regions, column_gap_px)

def pipeline_qwen2vl_2b_layout(image: np.ndarray, qwen_model, qwen_processor,
                                layout_model, config: dict) -> dict:
    ordered_regions = detect_and_order_regions(image, layout_model, config["layout_column_gap_px"])

    if not ordered_regions:
        # Layout model found nothing usable -- fall back to full-page OCR
        # rather than silently returning empty text.
        fallback = pipeline_qwen2vl_2b_raw(image, qwen_model, qwen_processor, config)
        fallback["regions"] = []
        fallback["fallback_to_full_page"] = True
        return fallback

    region_texts = []
    for region in ordered_regions:
        x1, y1, x2, y2 = [int(v) for v in region["bbox"]]
        x1, y1 = max(0, x1), max(0, y1)
        x2, y2 = min(image.shape[1], x2), min(image.shape[0], y2)
        if x2 <= x1 or y2 <= y1:
            continue
        crop = image[y1:y2, x1:x2]
        if region["type"] == "Figure":
            # Skip pure figures/ads for text OCR, per your OCR_PROMPT policy
            # of skipping non-text visual content -- but keep the region
            # logged so you can see what was excluded.
            region["ocr_skipped"] = True
            continue

            crop, qwen_model, qwen_processor, OCR_PROMPT_REGION,
            config["max_new_tokens"], config["max_image_longest_side"],
        )
        region["ocr_skipped"] = False
        region_texts.append(crop_text)

    combined_text = "\n\n".join(region_texts)
    return {"text": combined_text, "regions": ordered_regions, "fallback_to_full_page": False}

```python
def _render_page_image(pdf_path: Path, page_index: int, doc_id: str, dpi: int) -> str:
    """Render one page (0-indexed) to PNG. Cached: skips re-render if the
    file already exists, so re-running load_pages() is cheap."""
    out_dir = Path(RENDERED_IMAGE_DIR) / doc_id
    out_dir.mkdir(parents=True, exist_ok=True)

    pix.save(str(out_path))

    is rendered once to data/interim/page_images/<doc_id>/page_NNNN.png
    and returned as Page(id="<doc_id>_pNNNN", image_path=..., doc_id=...).
    Page number is NOT a separate contract field -- it's encoded in `id`
    (1-indexed, zero-padded to 4 digits); downstream stages needing the
    number parse it back out of `id`.
    """
    raw_dir = Path(cfg.get("raw_data_dir", "data/raw"))
    dpi = cfg.get("dpi", DEFAULT_DPI)

    pdf_paths = sorted(raw_dir.glob("*.pdf"))
    if not pdf_paths:
        raise FileNotFoundError(
            f"No PDFs found in {raw_dir} -- did scripts/get_data.sh run successfully?"
        )

    pages: list[Page] = []
    for pdf_path in pdf_paths:
        doc_id = pdf_path.stem
        doc = fitz.open(pdf_path)
        num_pages = doc.page_count
        doc.close()

        for page_index in range(num_pages):
            image_path = _render_page_image(pdf_path, page_index, doc_id, dpi)
            page_id = f"{doc_id}_p{page_index + 1:04d}"
            pages.append(Page(id=page_id, image_path=image_path, doc_id=doc_id))

    return pages
```

```
        if not images:
            break
```

```
        images[0].save(image_path, "PNG")
```

```python
        pages.append(
            Page(
                id=f"{doc_id}_p{page_number:04d}",
                image_path=str(image_path),
                doc_id=doc_id,
            )
        )
```

```python
        page_number += 1
```

```
    return pages
```

### 1. Deskew: probably unnecessary

If your scans are consistently straight, adding deskew doesn't buy you much.

Deskew is useful when:

* the physical page was scanned at an angle,
* text baselines are visibly tilted,
* OCR struggles because of rotation.

But your loader is already producing the page image as scanned. If the pages look straight, running a deskew algorithm over **hundreds/thousands of pages** adds computation and introduces another transformation that can potentially:

* slightly resample text,
* blur fine characters,
* crop edges incorrectly,
* misinterpret large graphics as text orientation.

So I'd make deskew **optional**, not mandatory.

# 2. Denoising: YES, but gently

Old magazine scans can have:

* paper texture,
* scanner noise,
* speckles,
* printing artifacts,
* bleed-through,
* compression artifacts.

```text
      ↓
from
      ↓
```

```text
scan

grayscale

denoise
 ↓
binarize

OCR
```

# The strongest argument FOR binarization

There *is* a legitimate reason to use it.

OCR models ultimately need to recognize characters. A good adaptive threshold can sometimes make:

```text
faded black text
+
yellow-ish paper
+
scanner noise
```

much easier to recognize.

For example:

```text
original
      ↓
grayscale
      ↓
adaptive threshold
      ↓
high-contrast text
```

could improve OCR on particularly poor scans.

So binarization isn't inherently bad.

The problem is making it **the canonical representation of your corpus**.

---

# The strongest argument AGAINST it

Your system isn't simply:

> image → OCR

Your project is:

> scanned page → layout understanding → OCR → chunks → embeddings → retrieval → grounded answer

And your `layout.detect()` stage comes **before OCR**:

```python

pages = enhance.run(pages, cfg)

text = ocr.transcribe(regions, cfg)
```

That's important.

A layout model may benefit from seeing:

> **the actual page**

rather than a heavily thresholded black-and-white representation.

A colored advertisement, a figure, a boxed code listing, a heading, and a normal paragraph may have useful visual distinctions.

If you binarize first, you potentially throw away those cues.

And there's another nasty problem:

### Colored text

Imagine:

```text
dark blue text
on
light blue background
```

A global threshold might decide:

```text
blue ≈ background
```

and make the text nearly disappear.

Or:

```text

on
yellow background
```

may have poor grayscale contrast even though the colors are visually very distinct.

So:

**binarization can improve some text OCR while simultaneously destroying information useful to layout detection and other visual understanding.**

That's a bad trade for your particular pipeline.

# 5. And what the hell does "augment" mean here? 😭

**Data augmentation** normally means artificially modifying training examples to create additional training data.

For example:

```text
original page

    ├── rotated 1°
    ├── slightly blurred
    ├── brightness changed
    ├── contrast changed
    └── cropped
```

You'd use those during **model training** to make the model robust.

For your inference/indexing pipeline, though, you generally **don't want to create five versions of every page**.

Otherwise:

```text

 ↓
OCR
 ↓
5× chunks
 ↓
5× embeddings

5× index entries
```

Now you've made your corpus larger and potentially introduced duplicate/redundant retrieval results.

For BYTE Retriever, that would be pretty pointless unless you specifically have a training stage where augmentation is needed.

So I'd say the `augment` comment in your stub is probably **too broad for the actual ingestion pipeline**.

```python
"""Stage 2 — layout detection / segmentation."""

from __future__ import annotations

from pathlib import Path

from PIL import Image

from ..contracts import Page, Region


# Map model-specific labels to our fixed Region contract.
#
# DocLayout-YOLO / DocStructBench has more classes than our contract,
# so several model classes are intentionally mapped to the closest
# supported Region kind.
_LABEL_MAP = {
    "title": "heading",
    "section-header": "heading",
    "plain_text": "text",
    "text": "text",
    "list-item": "text",
    "table": "table",
    "picture": "figure",
    "figure": "figure",
    "figure-caption": "text",
    "table-caption": "text",
    "page-header": "text",
    "page-footer": "text",
    "footnote": "text",
    "formula": "text",
}


_model = None


def _get_model(cfg: dict):
    """Lazy-load the DocLayout-YOLO model once."""

    global _model

    if _model is not None:
        return _model

    from doclayout_yolo import YOLOv10

    model_path = cfg.get("layout", {}).get(
        "model_path",
        "doclayout_yolo_docstructbench_imgsz1024.pt",
    )

    _model = YOLOv10(model_path)

    return _model


def _sort_reading_order(
    boxes: list[dict],
    page_width: int,
) -> list[dict]:
    """
    Sort detected regions into approximate multi-column reading order.

    BYTE commonly uses 2–3 running columns. We therefore group regions
    by their horizontal position first, then sort top-to-bottom inside
    each column.
    """

    if not boxes:
        return boxes

    # Tolerance for grouping regions into the same column.
    gap_px = 0.04 * page_width

    columns: list[list[dict]] = []

    for box in sorted(
        boxes,
        key=lambda b: b["bbox"][0],
    ):
        x0 = box["bbox"][0]
        placed = False

        for column in columns:
            column_x0 = min(
                b["bbox"][0]
                for b in column
            )

            if abs(x0 - column_x0) <= gap_px * 3:
                column.append(box)
                placed = True
                break

        if not placed:
            columns.append([box])

    # Left column → right column.
    columns.sort(
        key=lambda column: min(
            b["bbox"][0]
            for b in column
        )
    )

    # Top → bottom within each column.
    ordered: list[dict] = []

    for column in columns:
        column.sort(
            key=lambda b: b["bbox"][1]
        )
        ordered.extend(column)

    return ordered


def detect(
    pages: list[Page],
    cfg: dict,
) -> list[Region]:
    """
    Detect text/table/figure/heading regions.

    Returns regions in approximate BYTE multi-column reading order.
    """

    model = _get_model(cfg)

    regions: list[Region] = []

    score_thr = cfg.get("layout", {}).get(
        "score_thr",
        0.5,
    )

    for page in pages:

        image_path = Path(page.image_path)

        with Image.open(image_path) as image:
            image = image.convert("RGB")
            page_width, page_height = image.size

            # Run DocLayout-YOLO inference.
            results = model.predict(
                str(image_path),
                imgsz=1024,
                conf=score_thr,
                verbose=False,
            )

        boxes: list[dict] = []

        for result in results:

            if result.boxes is None:
                continue

            xyxy = result.boxes.xyxy.tolist()
            confidences = result.boxes.conf.tolist()
            classes = result.boxes.cls.tolist()

            for bbox, confidence, class_id in zip(
                xyxy,
                confidences,
                classes,
            ):
                confidence = float(confidence)

                if confidence < score_thr:
                    continue

                class_id = int(class_id)

                # Get model class name.
                if hasattr(result, "names"):
                    label = result.names[class_id]
                else:
                    label = str(class_id)

                label = label.lower().replace("_", "-")

                kind = _LABEL_MAP.get(
                    label,
                    "text",
                )

                x0, y0, x1, y1 = map(
                    int,
                    bbox,
                )

                # Clamp coordinates to page boundaries.
                x0 = max(0, min(x0, page_width))
                y0 = max(0, min(y0, page_height))
                x1 = max(0, min(x1, page_width))
                y1 = max(0, min(y1, page_height))

                if x1 <= x0 or y1 <= y0:
                    continue

                boxes.append(
                    {
                        "bbox": (x0, y0, x1, y1),
                        "kind": kind,
                        "score": confidence,
                    }
                )

        ordered_boxes = _sort_reading_order(
            boxes,
            page_width=page_width,
        )

        for box in ordered_boxes:
            regions.append(
                Region(
                    page_id=page.id,
                    bbox=box["bbox"],
                    kind=box["kind"],
                )
            )

    return regions
```

```python
ocr_path.write_text('''"""Stage 3 — OCR using pretrained TrOCR."""

from __future__ import annotations

import json
from pathlib import Path

import torch
from PIL import Image

from ..contracts import Page, Region, Chunk


# Saved OCR output.
OCR_DIR = Path(
    "/kaggle/working/doc-agent-G21/data/interim/ocr"
)


class Reader:
    """Pretrained TrOCR reader for text regions."""

    def __init__(self, cfg: dict) -> None:
        self.cfg = cfg["ocr"]

        model_name = self.cfg.get(
            "model",
            "microsoft/trocr-base-printed",
        )

        from transformers import (
            TrOCRProcessor,
            VisionEncoderDecoderModel,
        )

        self.device = (
            "cuda"
            if torch.cuda.is_available()
            else "cpu"
        )

        print(f"Loading OCR model: {model_name}")
        print(f"OCR device: {self.device}")

        self.processor = (
            TrOCRProcessor.from_pretrained(model_name)
        )

        self.model = (
            VisionEncoderDecoderModel.from_pretrained(
                model_name
            )
        )

        self.model.to(self.device)
        self.model.eval()

        print("OCR model loaded successfully.")

    def transcribe_region(
        self,
        region: Region,
        page_images: dict[str, str],
    ) -> str:
        """Crop one detected region and transcribe its text."""

        image_path = page_images.get(region.page_id)

        if image_path is None:
            raise KeyError(
                f"No image found for page_id={region.page_id}"
            )

        with Image.open(image_path) as image:
            image = image.convert("RGB")

            x0, y0, x1, y1 = region.bbox

            # Keep coordinates inside the image.
            x0 = max(0, x0)
            y0 = max(0, y0)
            x1 = min(image.width, x1)
            y1 = min(image.height, y1)

            if x1 <= x0 or y1 <= y0:
                return ""

            crop = image.crop(
                (x0, y0, x1, y1)
            )

            pixel_values = self.processor(
                images=crop,
                return_tensors="pt",
            ).pixel_values.to(self.device)

            with torch.no_grad():
                generated_ids = self.model.generate(
                    pixel_values,
                    max_new_tokens=512,
                )

            text = self.processor.batch_decode(
                generated_ids,
                skip_special_tokens=True,
            )[0]

        return text.strip()


def transcribe(
    regions: list[Region],
    cfg: dict,
    pages: list[Page],
) -> list[Chunk]:
    """Convert detected regions into OCR text chunks."""

    if not regions:
        return []

    # Map page IDs to their processed image paths.
    page_images = {
        page.id: page.image_path
        for page in pages
    }

    reader = Reader(cfg)

    chunks: list[Chunk] = []

    for index, region in enumerate(regions):

        text = reader.transcribe_region(
            region,
            page_images,
        )

        # Skip completely empty OCR results.
        if not text:
            continue

        chunks.append(
            Chunk(
                id=f"{region.page_id}_r{index:04d}",
                doc_id=pages[0].doc_id,
                text=text,
                page_ids=[region.page_id],
            )
        )

    # Save OCR results.
    OCR_DIR.mkdir(
        parents=True,
        exist_ok=True,
    )

    doc_id = pages[0].doc_id

    output_path = (
        OCR_DIR / f"{doc_id}.json"
    )

    with output_path.open(
        "w",
        encoding="utf-8",
    ) as f:
        json.dump(
            [
                chunk.model_dump()
                for chunk in chunks
            ],
            f,
            indent=2,
            ensure_ascii=False,
        )
```

```python
def transcribe(
    regions: list[Region],
    cfg: dict,
) -> list[Chunk]:
    """Convert layout regions into OCR text chunks.
    Checkpoints incrementally to OCR_DIR/chunks.json and resumes from it
    on restart (skips regions whose chunk_id was already OCR'd) -- Kaggle
    sessions can die mid-run, this avoids losing completed work."""

    OCR_DIR.mkdir(parents=True, exist_ok=True)
    output_path = OCR_DIR / "chunks.json"
    checkpoint_every = cfg["ocr"].get("checkpoint_every", 25)

    # Resume: load any chunks already OCR'd from a previous run.
    chunks: list[Chunk] = []
    already_done_ids: set[str] = set()
    if output_path.exists():
        with output_path.open("r", encoding="utf-8") as f:
            existing = json.load(f)
        chunks = [Chunk(**item) for item in existing]
        already_done_ids = {c.id for c in chunks}
        print(f"Resuming: {len(chunks)} chunks already OCR'd, skipping those.")

    reader = Reader(cfg)

    for index, region in enumerate(regions):
        doc_id = region.page_id.rsplit("_p", 1)[0]
        chunk_id = f"{region.page_id}_region_{index:04d}"

        if chunk_id in already_done_ids:
            continue  # already OCR'd in a previous (interrupted) run

        try:
            text = reader.transcribe_region(region)
        except Exception as exc:
            print(f"WARNING: OCR failed for region {index}/{len(regions)} ({chunk_id}): {exc}")
            continue

        if not text:
            continue

        chunks.append(Chunk(
            id=chunk_id, doc_id=doc_id, text=text,
            page_ids=[region.page_id], score=0.0,
        ))

        if (index + 1) % checkpoint_every == 0 or (index + 1) == len(regions):
            with output_path.open("w", encoding="utf-8") as f:
                json.dump([c.model_dump() for c in chunks], f, ensure_ascii=False, indent=2)
            print(f"[{index + 1}/{len(regions)}] checkpoint saved -- {len(chunks)} chunks total")

    return chunks
```

CHUNK_DIR = Path("data/interim/chunks")


def split(chunks: list[Chunk], cfg: dict) -> list[Chunk]:
    """Re-chunk raw per-region OCR chunks to cfg['index'] size/overlap.
    Input: small, region-sized Chunks from ocr.transcribe() (one per detected
    region — text/table/heading blocks), already in correct multi-column
    reading order from layout.py. This function regroups them, never reorders.
    """
    index_cfg = cfg.get("index", {})
    target_tokens = index_cfg.get("chunk_tokens", 256)
    overlap_tokens = index_cfg.get("overlap", 32)

    by_doc: dict[str, list[Chunk]] = {}
    for c in chunks:
        by_doc.setdefault(c.doc_id, []).append(c)

    output: list[Chunk] = []

    for doc_id, doc_chunks in by_doc.items():
        buffer_words: list[str] = []
        buffer_page_ids: list[str] = []
        chunk_idx = 0

        def flush():
            nonlocal chunk_idx
            if not buffer_words:
                return
            text = " ".join(buffer_words)
            seen = set()
            ordered_pages = [p for p in buffer_page_ids if not (p in seen or seen.add(p))]
            output.append(Chunk(
                id=f"{doc_id}_c{chunk_idx:04d}",
                doc_id=doc_id,
                text=text,
                page_ids=ordered_pages,
            ))
            chunk_idx += 1

        for c in doc_chunks:
            words = c.text.split()
            for w in words:
                buffer_words.append(w)
                if c.page_ids:
                    buffer_page_ids.extend(c.page_ids)

                if len(buffer_words) >= target_tokens:
                    flush()
                    carry = buffer_words[-overlap_tokens:] if overlap_tokens > 0 else []
                    carry_pages = buffer_page_ids[-1:] if buffer_page_ids else []
                    buffer_words = list(carry)
                    buffer_page_ids = list(carry_pages)

        flush()

    # Checkpoint, matching layout.py/ocr.py's pattern — lets embed.py (or you,
    # re-running this notebook later) reload without recomputing.
    CHUNK_DIR.mkdir(parents=True, exist_ok=True)
    out_path = CHUNK_DIR / "chunks.json"
    with out_path.open("w", encoding="utf-8") as f:
        json.dump([c.model_dump() for c in output], f, indent=2, ensure_ascii=False)
    print(f"Saved {len(output)} final chunks to {out_path}")

    return output
